function handleSaveAsset(event) { event.preventDefault(); // Ambil senarai aset sedia ada dari LocalStorage let assets = JSON.parse(localStorage.getItem('susut_nilai_assets')) || []; // Semak kategori jika pengguna pilih "Other Asset" let category = document.getElementById('asset-category').value; if (category === 'Other Asset') { category = document.getElementById('other-category-input').value || 'Other Asset'; } const unit = parseFloat(document.getElementById('unit').value) || 0; const unitPrice = parseFloat(document.getElementById('unit-price').value) || 0; // Bina objek aset baharu const newAsset = { id: Date.now(), purchaseDate: document.getElementById('purchase-date').value, disposalDate: '-', name: document.getElementById('asset-name').value, category: category, refNo: document.getElementById('invoice-number').value, model: 'Standard', usefulLife: document.getElementById('useful-life').value, unit: unit, unitPrice: unitPrice, total: unit * unitPrice, location: document.getElementById('department').value, updatedDate: new Date().toISOString().split('T')[0], updatedBy: document.getElementById('supplier-name').value }; // Tambah aset dan simpan semula assets.push(newAsset); localStorage.setItem('susut_nilai_assets', JSON.stringify(assets)); alert(`Asset "${newAsset.name}" successfully registered! Redirecting to Asset Register...`); window.location.href = "assetregister.html"; }
